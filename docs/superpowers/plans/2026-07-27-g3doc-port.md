```python
"""Source path -> g3doc path. Pure; no I/O. All layout rules live here."""
import re

PART_DIRS = {
    "part1-single-node": "Part I — Single node",
    "part2-inter-node": "Part II — Inter-node communication",
    "part3-clustering-execution": "Part III — Clustering & distributed execution",
    "part4-platform-reference-arch": "Part IV — Platform & reference architectures",
    "part5-operations-diagnostics": "Part V — Operations, diagnostics & troubleshooting",
    "part6-architecture-gcp-integration": "Part VI — Architecture & GCP integration",
}

# doc-15 is described in the overview as Part II's scaling bridge.
SPECIAL = {
    "README.md": "g3doc/index.md",
    "docs/00-guide-overview.md": "g3doc/overview.md",
    "docs/15-scaling-shape-of-the-cliff.md": "g3doc/part2-inter-node/scaling.md",
}

_NUM_PREFIX = re.compile(r"^\d+[-–—]\s*")


def _strip_num(name: str) -> str:
    """01-gpu-microarchitecture.md -> gpu-microarchitecture.md (T1-* untouched)."""
    return _NUM_PREFIX.sub("", name)


def src_to_g3doc(src_rel: str) -> str | None:
    src_rel = src_rel.lstrip("./")
    if src_rel in SPECIAL:
        return SPECIAL[src_rel]
    if not src_rel.endswith(".md"):
        return None
    if src_rel.startswith("docs/superpowers/"):
        return None
    if src_rel.startswith("labs/"):
        parts = src_rel.split("/")
        if len(parts) != 3 or parts[2] != "README.md":
            return None
        m = re.match(r"(lab-\d+)", parts[1])
        return f"g3doc/labs/{m.group(1)}/index.md" if m else None
    if src_rel.startswith("reference/"):
        return f"g3doc/reference/{src_rel.split('/', 1)[1]}"
    if src_rel.startswith("docs/toolkit/"):
        return f"g3doc/toolkit/{src_rel.split('/')[-1]}"
    if src_rel.startswith("docs/"):
        parts = src_rel.split("/")
        if len(parts) == 3 and parts[1] in PART_DIRS:
            return f"g3doc/{parts[1]}/{_strip_num(parts[2])}"
    return None


def page_slug(src_rel: str) -> str:
    """Stable, unique per page; used to name that page's diagram SVGs."""
    g = src_to_g3doc(src_rel)
    if g is None:
        raise ValueError(f"not a ported page: {src_rel}")
    rel = g[len("g3doc/"):]
    if rel.endswith("/index.md"):
        return rel[: -len("/index.md")].split("/")[-1]
    return rel.split("/")[-1][: -len(".md")]


LAB_DIRS: dict[str, str] = {}  # populated by build.py from the source tree
```

```python
"""Extract Mermaid blocks, derive alt text, render to SVG via mmdc."""
import json
import os
import re
import subprocess
import tempfile
from collections import namedtuple
from pathlib import Path

Block = namedtuple("Block", "start end code alt")

CHROME = "/usr/bin/google-chrome"
_FIG = re.compile(r"^\*\s*Figure\s*:\s*", re.I)


def _clean_caption(line: str) -> str:
    s = line.strip()
    s = _FIG.sub("", s)
    s = s.strip("*").strip()
    return s.rstrip(".")


def _alt_for(lines: list[str], fence_idx: int) -> str:
    """Italic caption directly above the fence; else nearest heading."""
    prev = [l for l in lines[max(0, fence_idx - 5):fence_idx] if l.strip()]
    if prev and prev[-1].lstrip().startswith("*"):
        cap = _clean_caption(prev[-1])
        if cap:
            return cap
    for l in reversed(lines[:fence_idx]):
        if l.startswith("#"):
            return f"diagram: {l.lstrip('#').strip()}"
    return "diagram"


def extract_blocks(md: str) -> list[Block]:
    lines = md.split("\n")
    out, i = [], 0
    while i < len(lines):
        if lines[i].strip().startswith("```mermaid"):
            j = i + 1
            while j < len(lines) and not lines[j].strip().startswith("```"):
                j += 1
            out.append(Block(i, j, "\n".join(lines[i + 1:j]), _alt_for(lines, i)))
            i = j + 1
        else:
            i += 1
    return out


def _pptr_config(tmp: Path) -> Path:
    """Puppeteer must use system Chrome; the bundled download fails on this VM."""
    p = tmp / "pptr.json"
    p.write_text(json.dumps({
        "executablePath": CHROME,
        "args": ["--no-sandbox", "--disable-dev-shm-usage"],
    }))
    return p


def render(code: str, out_svg: Path, mmdc: str, tmp: Path) -> None:
    mmd = tmp / "d.mmd"
    mmd.write_text(code)
    out_svg.parent.mkdir(parents=True, exist_ok=True)
    subprocess.run(
        [mmdc, "-i", str(mmd), "-o", str(out_svg), "-p", str(_pptr_config(tmp))],
        check=True, capture_output=True, text=True,
    )
    if not out_svg.exists() or out_svg.stat().st_size == 0:
        raise RuntimeError(f"empty SVG for {out_svg}")
```

IMG = re.compile(r"\!\[([^\]]*)\]\(([^)]+)\)")

# Broken on main: real file is 20-performance-monitoring-day2-ops.md (9 refs).

def _lab_dir_to_page(target: str) -> str | None:
    m = re.search(r"labs/(lab-\d+)[^/]*/?$", target)
    return f"labs/{m.group(1)}/index.md" if m else None


def _resolve(src_rel: str, target: str) -> str | None:
    """Map a link target from source-tree space into g3doc space."""
    for bad, good in PRE_FIX.items():
        target = target.replace(bad, good)
    clean, _, anchor = target.partition("#")
    if not clean:
        return None
    src_abs = posixpath.normpath(posixpath.join(posixpath.dirname(src_rel), clean))

    if src_abs.startswith("assets/"):
        new_abs = src_abs
    elif (lab := _lab_dir_to_page(src_abs)) and not src_abs.endswith(".md"):
        new_abs = lab
    else:
        g = src_to_g3doc(src_abs if src_abs.endswith(".md")
                         else posixpath.join(src_abs, "README.md"))
        if g is None:
            return None
        new_abs = g[len("g3doc/"):]

    here = src_to_g3doc(src_rel)
    if here is None:
        return None
    new_rel = posixpath.relpath(new_abs, posixpath.dirname(here[len("g3doc/"):]))
    return new_rel + (f"#{anchor}" if anchor else "")


def rewrite_links(md: str, src_rel: str) -> str:
    def sub(m: re.Match) -> str:
        text, target = m.group(1), m.group(2).strip()
        if target.startswith(SKIP):
            return m.group(0)
        new = _resolve(src_rel, target)
        return m.group(0) if new is None else f"[{text}]({new})"

    def sub_img(m: re.Match) -> str:
        text, target = m.group(1), m.group(2).strip()
        if target.startswith(SKIP):
            return m.group(0)
        new = _resolve(src_rel, target)
        return m.group(0) if new is None else f"![{text}]({new})"

    return LINK.sub(sub, IMG.sub(sub_img, md))
```

```python
"""Orchestrator: src/ -> g3doc/. Replaces Mermaid blocks with rendered SVGs."""
import posixpath
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
from diagrams import extract_blocks          # noqa: E402
from links import rewrite_links              # noqa: E402
from pathmap import page_slug, src_to_g3doc  # noqa: E402


def _swap_diagrams(md: str, src_rel: str) -> str:
    """Replace each ```mermaid block with an image ref to its rendered SVG."""
    blocks = extract_blocks(md)
    if not blocks:
        return md
    lines = md.split("\n")
    out, prev, slug = [], 0, page_slug(src_rel)
    here = src_to_g3doc(src_rel)[len("g3doc/"):]
    img_rel = posixpath.relpath("images", posixpath.dirname(here))
    for i, b in enumerate(blocks, 1):
        out.extend(lines[prev:b.start])
        out.append(f"![{b.alt}]({img_rel}/{slug}-{i}.svg)")
        prev = b.end + 1
    out.extend(lines[prev:])
    return "\n".join(out)


def build_pages() -> int:
    n = 0
    for p in sorted(Path("src").rglob("*.md")):
        rel = str(p.relative_to("src"))
        dest = src_to_g3doc(rel)
        if dest is None:
            continue
        md = _swap_diagrams(p.read_text(), rel)
        md = rewrite_links(md, rel)
        out = Path(dest)
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_text(md)
        n += 1
    return n


if __name__ == "__main__":
    print("pages written:", build_pages())

"""Emit _toc.yaml and the directory index.md pages."""
import re
import sys
from pathlib import Path

import yaml

sys.path.insert(0, str(Path(__file__).parent))
from pathmap import PART_DIRS, src_to_g3doc  # noqa: E402

ROOT = "/g3doc"
SECTIONS = [
    ("Toolkit", "toolkit", "Cross-cutting NVIDIA tooling, T1-T6."),
    ("Labs", "labs", "21 labs, executed live on A3 H100 clusters."),
    ("Reference", "reference", "Glossary, XID table, driver matrix, tunables."),
]


def _title_of(page: Path) -> str:
    for line in page.read_text().split("\n"):
        if line.startswith("# "):
            return line[2:].strip()
    return page.stem


def _src_order(src_rel: str) -> tuple:
    """Order by the numeric prefix in the SOURCE name, not the stripped name."""
    name = src_rel.split("/")[-1]
    m = re.match(r"(?:T)?(\d+)", name if name != "README.md" else src_rel.split("/")[-2].replace("lab-", ""))
    return (int(m.group(1)) if m else 999, src_rel)


def _pages_for(subdir: str) -> list[str]:
    """g3doc-relative page paths under subdir, in source-number order."""
    pairs = []
    for p in Path("src").rglob("*.md"):
        rel = str(p.relative_to("src"))
        dest = src_to_g3doc(rel)
        if dest and dest.startswith(f"g3doc/{subdir}/"):
            pairs.append((_src_order(rel), dest))
    return [d for _, d in sorted(pairs)]


def build_toc() -> dict:
    entries = [{"title": "Overview", "path": f"{ROOT}/overview.md"}]
    for d, title in PART_DIRS.items():
        section = [{"title": "Overview", "path": f"{ROOT}/{d}/index.md"}]
        for dest in _pages_for(d):
            section.append({"title": _title_of(Path(dest)), "path": f"{ROOT}/{dest[len('g3doc/'):]}"})
        entries.append({"title": title, "section": section})
    for title, sub, _ in SECTIONS:
        section = [{"title": "Overview", "path": f"{ROOT}/{sub}/index.md"}]
        for dest in _pages_for(sub):
            section.append({"title": _title_of(Path(dest)), "path": f"{ROOT}/{dest[len('g3doc/'):]}"})
        entries.append({"title": title, "section": section})
    return {"toc": entries}


def write_indexes() -> int:
    """One index.md per part/section dir, listing its children."""
    n = 0
    groups = [(t, d) for d, t in PART_DIRS.items()] + [(t, s) for t, s, _ in SECTIONS]
    blurbs = {s: b for t, s, b in SECTIONS}
    for title, sub in groups:
        rows = []
        for dest in _pages_for(sub):
            p = Path(dest)
            name = p.parent.name if p.name == "index.md" else p.name
            rows.append(f"- [{_title_of(p)}]({name if p.name != 'index.md' else name + '/index.md'})")
        body = [f"# {title}", ""]
        if sub in blurbs:
            body += [blurbs[sub], ""]
        body += rows + [""]
        out = Path(f"g3doc/{sub}/index.md")
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_text("\n".join(body))
        n += 1
    return n


if __name__ == "__main__":
    print("indexes:", write_indexes())
    Path("g3doc/_toc.yaml").write_text(
        yaml.safe_dump(build_toc(), sort_keys=False, allow_unicode=True, width=100)
    )
    print("wrote g3doc/_toc.yaml")
```

```python
"""Assert every relative link and image target under g3doc/ resolves."""
import re
import sys
from pathlib import Path

TARGET = re.compile(r"\!?\[[^\]]*\]\(([^)]+)\)")
SKIP = ("http://", "https://", "mailto:", "#", "go/")


def check(root: str = "g3doc") -> list[str]:
    failures = []
    for md in sorted(Path(root).rglob("*.md")):
        for n, line in enumerate(md.read_text().split("\n"), 1):
            for target in TARGET.findall(line):
                t = target.strip()
                if t.startswith(SKIP):
                    continue
                path = t.split("#")[0]
                if not path:
                    continue
                if not (md.parent / path).exists():
                    failures.append(f"{md}:{n} -> {t}")
    return failures


if __name__ == "__main__":
    bad = check(sys.argv[1] if len(sys.argv) > 1 else "g3doc")
    for f in bad:
        print("BROKEN", f)
    print(f"\n{len(bad)} broken link(s)")
    sys.exit(1 if bad else 0)
```
